// ---------------------------------------------------------------- caustics: the light's beams
// The sun's and the lamps' light traced off and through HDRP's water (pt_caustics.hlsl) is kept as beams: each a triangle of the
// water and the three rays its corners sent on (reflected or refracted), a prism the light through that triangle fills. A point
// inside it takes its light: the light through the water's triangle over the prism's cross-section there (rays that close in
// make it small and the light a bright line; rays that spread, dim), from the way the rays run there, if one ray from the point
// back to its place on the water is clear (what stands between throws its shadow; what the three rays never met, a ledge before
// the wall they hit, is lit as it should be and shades what lies behind it). So a surface takes the light whatever its shape:
// nothing is laid flat between the three points the rays hit ("Interactive Rendering of Caustics using Interpolated Warped
// Volumes", Ernst, Akenine-Moller and Jensen, 2005, is the method; the first build stored the flat landed triangle, which put
// light on what lay near its plane and none on the rock beside it, and left out every triangle across an edge).
// A beam counts along its rays from a little before the nearest thing its rays met to a little past the farthest (its slab):
// that part's box is what the search finds (an acceleration structure of the boxes, built every frame: tracer.cpp; the
// hardware's search gives a point the beams round it alone, where the hash it replaces walked 8 buckets' lists at every level
// for each point: 40 to 50 ms of the lake's composite in game, run 20261007-140305).
// A beam (CAUSTIC_BEAM bytes, six float4s): its water triangle's corners (relative to the scene origin) with the slab's start,
// its end and its kind (bit 0: refracted, the water's absorption on its way; bits 1-2: the slot), then a point on each corner's
// ray (where it landed; along it for one that met nothing). Its light (CAUSTIC_LIGHT bytes): at each corner the light through
// the triangle that corner's ray carries (flux / pi in storage units, to_storage, before the fog or the water on its way: those
// are taken over the point's own distance).
static const uint CAUSTIC_NONE = 0xFFFFFFFFu;
static const uint CAUSTIC_BEAM = 96;
static const uint CAUSTIC_LIGHT = 48;
static const uint CAUSTIC_BOX = 32;
// A pixel or two where the camera sees a point: what a beam's slab reaches past its rays' ends at least, and how fine the
// splits at an edge go (pt_caustics).
float caustic_reach(float3 p) { return 0.005 + 0.002 * length(p - g.cameraPos.xyz); }
// The caustics' light at a surface point p (relative to the scene origin; n its triangle's normal), as any other light there: its
// irradiance as the shading normal takes it (diffuse: times albedo / pi) and, with a lobe (alpha > 0), what its GGX lobe seen
// from v reflects: every beam's light comes along its rays, so a normal map turns it as it turns the sun's, and a glossy or
// metal surface shows it as a highlight. Every ray that meets a surface takes it there (direct_light: path vertices, what
// reflections show; the camera's pixels, pt_compose). In light units; none with the Caustics row off or no water.
void caustic_irradiance(float3 p, float3 n, float3 shading, float3 f0, float alpha, float3 v, out float3 diffuse, out float3 specular) {
    diffuse = 0.0; specular = 0.0;
    if (g.caustics.x == INVALID) return;
    RaytracingAccelerationStructure boxes = ResourceDescriptorHeap[g.caustics.y];
    RWByteAddressBuffer light = ResourceDescriptorHeap[g.caustics.x];
    RWByteAddressBuffer beams = ResourceDescriptorHeap[asuint(g.causticGrid.y)];
    RayDesc ray; ray.Origin = p; ray.Direction = n; ray.TMin = 0.0; ray.TMax = 1e-4;
    RayQuery<RAY_FLAG_SKIP_TRIANGLES> q;
    q.TraceRayInline(boxes, RAY_FLAG_NONE, 0xFF, ray);
    const float offset = 0.01 + 0.001 * length(p - g.cameraPos.xyz);
    while (q.Proceed()) {
        if (q.CandidateType() != CANDIDATE_PROCEDURAL_PRIMITIVE) continue;
        const uint i = q.CandidatePrimitiveIndex();
        const uint base = i * CAUSTIC_BEAM;
        const uint4 r0 = beams.Load4(base), r1 = beams.Load4(base + 16), r2 = beams.Load4(base + 32);
        const float3 xa = asfloat(r0.xyz), xb = asfloat(r1.xyz), xc = asfloat(r2.xyz);
        float3 da = asfloat(beams.Load3(base + 48)) - xa, db = asfloat(beams.Load3(base + 64)) - xb, dc = asfloat(beams.Load3(base + 80)) - xc;
        da = normalize(da); db = normalize(db); dc = normalize(dc);
        // How far along the three rays their triangle holds the point: where it lies in the plane of the three points that far
        // along (a cubic in the distance; Newton from the point's distance along the rays).
        float s = (dot(p - xa, da) + dot(p - xb, db) + dot(p - xc, dc)) / 3.0;
        float3 a, b, c, nt;
        [unroll] for (uint it = 0; it < 4; ++it) {
            a = xa + da * s; b = xb + db * s; c = xc + dc * s;
            nt = cross(b - a, c - a);
            if (it == 3) break;
            const float f = dot(nt, p - a);
            const float slope = dot(cross(db - da, c - a) + cross(b - a, dc - da), p - a) - dot(nt, da);
            if (abs(slope) > 1e-20) s -= f / slope;
        }
        if (!(s >= asfloat(r0.w) && s <= asfloat(r1.w))) continue;
        const float nn = dot(nt, nt);
        if (!(nn > 1e-24)) continue;
        // Inside the three rays there (its weights of b and c).
        const float wb = dot(cross(p - a, c - a), nt) / nn, wc = dot(cross(b - a, p - a), nt) / nn;
        if (!(wb >= 0.0 && wc >= 0.0 && wb + wc <= 1.0)) continue;
        const float wa = 1.0 - wb - wc;
        const float3 along = normalize(da * wa + db * wb + dc * wc); // the way its light runs there
        const float3 l = -along;
        if (!(dot(n, l) > 0.0)) continue;
        // The light through the water's triangle, over the beam's cross-section here: the irradiance across its rays.
        const float across = 0.5 * abs(dot(nt, along));
        if (!(across > 1e-16)) continue;
        const uint at = i * CAUSTIC_LIGHT;
        float3 e = (asfloat(light.Load3(at)) * wa + asfloat(light.Load3(at + 12)) * wb + asfloat(light.Load3(at + 24)) * wc) * (reference_radiance() * PI / across);
        if (!any(e > 0.0)) continue;
        // Its way here: the water it passed (a refracted beam) or HDRP's fog (a reflected one), over this point's own distance,
        // and one ray back to its place on the water for what stands between.
        const float3 x = xa * wa + xb * wb + xc * wc;
        const float3 back = x - p;
        const float dist = length(back);
        const uint kind = r2.w;
        e *= (kind & 1u) != 0u ? exp(-water_row((kind >> 1) & 3u, 14).xyz * dist) : exp(-height_fog_depth(x, p));
        const bool passed = g_passWater;
        g_passWater = true; // it ends on the water's waves: the level water under them is none of its business
        e *= transmittance(p + n * offset, back / max(dist, 1e-6), 0.0, max(dist - offset - 0.01, 0.0));
        g_passWater = passed;
        diffuse += e * max(dot(shading, l), 0.0);
        if (alpha > 0.0) specular += e * ggx_specular(shading, v, l, f0, alpha);
    }
}
// Its diffuse share alone on the plane of the point's triangle: irradiance / pi (a diffuse surface's radiance per albedo).
float3 caustic_light(float3 p, float3 n) {
    float3 diffuse, specular;
    caustic_irradiance(p, n, n, 0.0, 0.0, n, diffuse, specular);
    return diffuse / PI;
}
