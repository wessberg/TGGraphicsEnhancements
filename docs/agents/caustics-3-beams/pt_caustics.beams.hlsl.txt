// Caustics (the Caustics row, PT-14): the sun's and the lamps' light traced onto HDRP's water as its waves shape it, reflected
// and refracted there and traced on to what it meets, kept as beams any ray can ask at the surface it meets (caustic_irradiance
// in pt_common.hlsli). The method is beam tracing from the light, as in "Interactive Rendering of Caustics using Interpolated
// Warped Volumes" (Ernst, Akenine-Moller and Jensen, 2005) and the water caustics of "Real-Time Ray Traced Caustics" (Ray Tracing
// Gems II, chapter 30): a grid of rays over the water, each cell's light carried between its corners' rays. Where those rays
// close in (a wave focusing the light) the beam is narrow and its light a bright line; where they spread, dim. No blur and no
// noise come from the method.
//   - The grid: levels of 512 x 512 cells round the camera (tracer.cpp: 2 cm the finest, each next twice as coarse, holding the
//     finer one's square in its middle), moved by a random fraction of a cell every frame. A group traces a tile of 15 x 15
//     cells (its 16 x 16 corners) of one level over one slot's water.
//   - A corner: the water's level surface under it (a ray down onto the slot's water instance), then HDRP's waves there as HDRP's
//     water shader moves its surface (water_point): each band's displacement (its height, and its horizontal move times HDRP's
//     choppiness of 2.25) and surface gradient (read bicubic, as HDRP reads it), times its amplitude and its fade with the
//     distance from the camera.
//   - The lights of a tile: the two that send its water the most light, and one of all the others, chosen in proportion to its
//     light and divided by its chance (every lamp counts; its noise is the weak lamps' alone). A lamp's light leaves a random
//     point of its emitter's ball (the same for every tile of the frame), the sun's a random direction within its disc.
//   - Each corner's light: a shadow ray to the light (glass tints it), and two rays on from the wave: the reflection and the
//     refraction (an index of 1.333, the exact Fresnel split between them), each to the first thing it meets.
//   - Each cell, two triangles, each a beam per branch: the water's triangle, its three rays, and the light through it (a
//     lamp's intensity times the solid angle the triangle fills seen from it; the sun's irradiance times the triangle's area
//     across its light) as the corners carry it. A point between the three rays takes that light over the beam's cross-section
//     there, if a ray from it back to the water is clear (caustic_irradiance): on whatever surface it lies. No beam is left out
//     for where its rays went (one across an edge lights both sides, one with a ray into the sky lights what the others met).
//     Tiles none of whose rays land in or near the camera's view go in cells of 5 x 5: the same light in fewer, wider beams.
//   The beams' boxes go into a buffer an acceleration structure is built over every frame (tracer.cpp).
//   pass.a.x: 0 empties the counters, 1 traces (SV_GroupID.z = level x 4 + slot), 2 makes the room past this frame's beams
//   inactive (a thread a beam); pass.a.y: the boxes (raw UAV)
#include "pt_common.hlsli"

static const uint TILE = 16;            // corners a tile's side
static const uint CELLS = 15;           // cells a tile's side
static const uint GRID = 512;           // cells a level's side (tracer.cpp kCausticGrid)
static const uint SLOTS = 4;            // TGPT_MAX_WATER
static const float WATER_IOR = 1.3333;  // HDRP's
static const float CHOPPINESS = 2.25;   // HDRP's WATER_SYSTEM_CHOPPINESS
static const uint SUN = 0xFFFFFFFEu;
static const uint PICKS = 3;

// A level's cell (the finest, doubled per level).
float caustic_cell(uint level) { return g.causticGrid.x * exp2(float(level)); }

groupshared uint gs_light[PICKS];        // the tile's lights (SUN, a light's index, CAUSTIC_NONE)
groupshared float gs_scale[PICKS];       // and what each one's light counts for
groupshared float3 gs_water[TILE * TILE];    // each corner's point on the waves (relative to the scene origin)
groupshared float3 gs_land[2][TILE * TILE];  // a point on its reflected [0] and refracted [1] rays: where each landed, or 1 m along
groupshared uint gs_state[2][TILE * TILE];   // 0: no ray leaves the corner, 1: its ray met nothing (or carries none), 2: it landed
groupshared uint gs_seen[2];                 // whether any of the tile's rays of a branch landed in or near the camera's view
groupshared float3 gs_carried[2][TILE * TILE]; // the light each carries as it leaves the wave: the light's intensity (a lamp's,
                                               // per steradian) or irradiance (the sun's), times its share and the Fresnel split

// HDRP's waves read as HDRP's water shader reads its surface gradient: a cubic B-spline over 4 x 4 texels, as four bilinear reads.
float4 bicubic(Texture2DArray<float4> t, float2 uv, float slice, float size) {
    const float2 st = uv * size - 0.5;
    const float2 i = floor(st), f = st - i, f2 = f * f, f3 = f2 * f;
    const float2 w0 = (1.0 - 3.0 * f + 3.0 * f2 - f3) / 6.0, w1 = (4.0 - 6.0 * f2 + 3.0 * f3) / 6.0;
    const float2 w2 = (1.0 + 3.0 * f + 3.0 * f2 - 3.0 * f3) / 6.0, w3 = f3 / 6.0;
    const float2 g0 = w0 + w1, g1 = w2 + w3;
    const float2 h0 = (i - 0.5 + w1 / g0) / size, h1 = (i + 1.5 + w3 / g1) / size;
    return g0.y * (g0.x * t.SampleLevel(s_linearRepeat, float3(h0.x, h0.y, slice), 0) + g1.x * t.SampleLevel(s_linearRepeat, float3(h1.x, h0.y, slice), 0)) +
           g1.y * (g0.x * t.SampleLevel(s_linearRepeat, float3(h0.x, h1.y, slice), 0) + g1.x * t.SampleLevel(s_linearRepeat, float3(h1.x, h1.y, slice), 0));
}

// The water of a slot at a point of the grid (x, z relative to the scene origin): its level surface below (the slot's own water
// instance, under any other slot's), moved as HDRP's waves move it there, and its normal. False where the slot has no water.
bool water_point(uint slot, float2 xz, out float3 position, out float3 normal) {
    position = float3(xz.x, 0.0, xz.y); normal = float3(0.0, 1.0, 0.0);
    StructuredBuffer<InstanceRecord> instances = ResourceDescriptorHeap[g.scene0.x];
    const float top = g.waterTop[slot], bottom = g.waterBottom[slot];
    RayDesc ray; ray.Origin = float3(xz.x, top + 0.05, xz.y); ray.Direction = float3(0.0, -1.0, 0.0); ray.TMin = 0.0; ray.TMax = top - bottom + 0.1;
    float3 base = 0.0, baseNormal = 0.0;
    bool found = false;
    for (uint pass = 0; pass < 4 && !found; ++pass) {
        PT_RAYQUERY(RAY_FLAG_SKIP_PROCEDURAL_PRIMITIVES | RAY_FLAG_FORCE_OPAQUE) q;
        q.TraceRayInline(g_scene, RAY_FLAG_NONE, MASK_WATER, ray);
        q.Proceed();
        if (q.CommittedStatus() != COMMITTED_TRIANGLE_HIT) return false;
        const float t = q.CommittedRayT();
        if (((instances[q.CommittedInstanceID()].flags >> 8) & 3u) != slot) { ray.TMin = t + 1e-3; continue; }
        base = ray.Origin + ray.Direction * t;
        baseNormal = fetch_hit(q.CommittedInstanceID(), q.CommittedGeometryIndex(), q.CommittedPrimitiveIndex(), q.CommittedTriangleBarycentrics(),
                               q.CommittedObjectToWorld3x4(), ray.Direction, base).normal;
        found = true;
    }
    if (!found || baseNormal.y < 0.5) return false; // none, or a waterfall's sheet
    // HDRP samples its waves in the surface's own space (rotated as the surface is; the world itself for a custom mesh's).
    const float3 r0 = water_row(slot, 9).xyz, r1 = water_row(slot, 10).xyz, r2 = water_row(slot, 11).xyz;
    float3 n = float3(dot(r0, baseNormal), dot(r1, baseNormal), dot(r2, baseNormal));
    float3 move = 0.0;
    const float4 info = water_row(slot, 12);
    const uint bands = uint(info.x);
    if (bands > 0u && g.waterTextures[slot] != INVALID && g.waterNormals[slot] != INVALID) {
        Texture2DArray<float4> displacement = ResourceDescriptorHeap[NonUniformResourceIndex(g.waterTextures[slot])];
        Texture2DArray<float4> gradient = ResourceDescriptorHeap[NonUniformResourceIndex(g.waterNormals[slot])];
        const float distance = length(base - g.cameraPos.xyz);
        float2 slope = 0.0;
        for (uint b = 0; b < bands; ++b) {
            const float4 rx = water_row(slot, b * 3), rz = water_row(slot, b * 3 + 1), e = water_row(slot, b * 3 + 2);
            const float2 uv = float2(dot(rx.xyz, base) + rx.w, dot(rz.xyz, base) + rz.w);
            // HDRP's fade with the distance from the camera: saturated, squared, then smoothed.
            float fade = saturate(distance * e.y + e.z); fade *= fade; fade = fade * fade * (3.0 - 2.0 * fade);
            if (!(fade > 0.0)) continue;
            const float3 raw = displacement.SampleLevel(s_linearRepeat, float3(uv, float(b)), 0).xyz * (e.x * fade);
            move += float3(-CHOPPINESS * raw.y, raw.x, -CHOPPINESS * raw.z);
            slope += bicubic(gradient, uv, float(b), info.y).xy * fade;
        }
        n = n - float3(slope.x, 0.0, slope.y);
    }
    n = normalize(n);
    position = base + r0 * move.x + r1 * move.y + r2 * move.z;
    normal = normalize(r0 * n.x + r1 * n.y + r2 * n.z);
    return all(isfinite(position)) && all(isfinite(normal));
}

// A point of a lamp's emitter for this frame (the same for every tile): uniform in its ball.
float3 light_point(LightRecord l, uint index) {
    Rng r; r.state = pcg(asuint(g.causticGrid.w) ^ pcg(index + 0x632BE5ABu));
    const float z = 1.0 - 2.0 * r.next(), phi = 2.0 * PI * r.next();
    const float s = sqrt(max(0.0, 1.0 - z * z)), radius = l.radius * pow(r.next(), 1.0 / 3.0);
    return l.position + float3(s * cos(phi), z, s * sin(phi)) * radius;
}
// The sun's direction for this frame: a point of its disc.
float3 sun_direction() {
    Rng r; r.state = pcg(asuint(g.causticGrid.w) ^ 0xA511E9B3u);
    return cone_sample(g.sunDir.xyz, g.sunDir.w, r.next2());
}

// How much light a light sends onto a tile's water at most (its box: the tile's square, the water's heights with the waves'
// largest move): the sun's irradiance times the square's area across it; a lamp's intensity times the square's area over the
// nearest distance squared (at most a hemisphere), none under the water nor out of the range of a light that keeps one. Never 0
// where it lights any.
float light_weight(uint li, float2 lo, float2 hi, float yLo, float yHi) {
    const float area = (hi.x - lo.x) * (hi.y - lo.y);
    if (li == SUN) return g.sunColor.w > 0.0 && g.sunDir.y > 0.0 ? luminance(g.sunColor.rgb) * area * g.sunDir.y : 0.0;
    const LightRecord l = load_light(li);
    if (l.type > 1u || l.position.y < yLo) return 0.0;
    const float3 c = clamp(l.position, float3(lo.x, yLo, lo.y), float3(hi.x, yHi, hi.y));
    const float d2 = dot(l.position - c, l.position - c);
    // A light that falls off with distance alone (LIGHT_NO_RANGE, PT-50: the plugin's lights) reaches the water however far it is;
    // its range is only how far the light grid lists it. One that keeps HDRP's window stops at its range.
    if ((l.flags & LIGHT_NO_RANGE) == 0u && !(d2 <= l.range * l.range)) return 0.0;
    return luminance(l.color) * min(area / max(d2, 1e-4), 2.0 * PI);
}
// The tile's lights, all 256 threads over all the lights (the sun and every lamp; a scene holds a thousand near the camera): the
// two strongest, and one of the rest in proportion to its weight, which then counts for all of the rest (their summed weight over
// its own). Each thread keeps its own share's two strongest and sum, then its share's pick of the rest (a weighted reservoir);
// one thread joins them (a share's pick taken by the share's sum: a light's chance is its weight over the rest's). The corners'
// light arrays (gs_carried) hold the shares meanwhile.
void pick_lights(uint index, uint slot, float2 lo, float2 hi, uint tile) {
    const float margin = water_row(slot, 12).z;
    const float yLo = g.waterBottom[slot] - margin, yHi = g.waterTop[slot] + margin;
    const uint count = g.settings2.z;
    Rng r; r.state = pcg(asuint(g.causticGrid.w) ^ pcg(tile * 0x9E3779B9u + index * 0x85EBCA6Bu + 0x7F4A7C15u));
    uint i0 = CAUSTIC_NONE, i1 = CAUSTIC_NONE; float w0 = 0.0, w1 = 0.0, total = 0.0;
    for (uint k = index; k <= count; k += TILE * TILE) {
        const uint li = k == count ? SUN : k;
        const float w = light_weight(li, lo, hi, yLo, yHi);
        if (!(w > 0.0)) continue;
        total += w;
        if (w > w0) { i1 = i0; w1 = w0; i0 = li; w0 = w; } else if (w > w1) { i1 = li; w1 = w; }
    }
    gs_carried[0][index] = float3(w0, w1, total);
    gs_carried[1][index] = float3(asfloat(i0), asfloat(i1), 0.0);
    GroupMemoryBarrierWithGroupSync();
    if (index == 0u) {
        uint a = CAUSTIC_NONE, b = CAUSTIC_NONE; float wa = 0.0, wb = 0.0, sum = 0.0;
        for (uint t = 0; t < TILE * TILE; ++t) {
            const float3 w = gs_carried[0][t];
            const uint2 ids = asuint(gs_carried[1][t].xy);
            sum += w.z;
            [unroll] for (uint j = 0; j < 2; ++j) {
                const float wj = j == 0 ? w.x : w.y; const uint ij = j == 0 ? ids.x : ids.y;
                if (!(wj > 0.0)) continue;
                if (wj > wa) { b = a; wb = wa; a = ij; wa = wj; } else if (wj > wb) { b = ij; wb = wj; }
            }
        }
        gs_light[0] = a; gs_scale[0] = 1.0;
        gs_light[1] = b; gs_scale[1] = 1.0;
        gs_light[2] = CAUSTIC_NONE; gs_scale[2] = sum - wa - wb; // the rest's weight, until its pick is made
    }
    GroupMemoryBarrierWithGroupSync();
    const uint a = gs_light[0], b = gs_light[1];
    float rest = 0.0, pickWeight = 0.0; uint pick = CAUSTIC_NONE;
    for (uint k2 = index; k2 <= count; k2 += TILE * TILE) {
        const uint li = k2 == count ? SUN : k2;
        if (li == a || li == b) continue;
        const float w = light_weight(li, lo, hi, yLo, yHi);
        if (!(w > 0.0)) continue;
        rest += w;
        if (r.next() * rest < w) { pick = li; pickWeight = w; }
    }
    gs_carried[0][index] = float3(rest, pickWeight, asfloat(pick));
    GroupMemoryBarrierWithGroupSync();
    if (index == 0u) {
        const float all = gs_scale[2];
        float sum = 0.0, chosenWeight = 0.0; uint chosen = CAUSTIC_NONE;
        for (uint t = 0; t < TILE * TILE; ++t) {
            const float3 s = gs_carried[0][t];
            if (!(s.x > 0.0)) continue;
            sum += s.x;
            if (r.next() * sum < s.x) { chosen = asuint(s.z); chosenWeight = s.y; }
        }
        if (chosen != CAUSTIC_NONE && sum > 1e-6 * (sum + all) && chosenWeight > 0.0) { gs_light[2] = chosen; gs_scale[2] = sum / chosenWeight; }
        else { gs_light[2] = CAUSTIC_NONE; gs_scale[2] = 0.0; }
    }
    GroupMemoryBarrierWithGroupSync();
}

// The light li arriving at a water point x: its direction, its intensity (a lamp's per steradian: the solid angle the water's
// triangle fills multiplies it, emit; the sun's irradiance) and how far its shadow ray goes.
void light_at(uint li, float3 x, out float3 toLight, out float3 intensity, out float tmax) {
    if (li == SUN) {
        toLight = sun_direction(); tmax = g.params.y;
        intensity = g.sunColor.rgb * sun_fog(x);
    } else {
        LightRecord l = load_light(li);
        l.position = light_point(l, li);
        float dist;
        intensity = light_incident(l, x, toLight, dist);
        intensity *= dist * dist;
        tmax = max(dist - light_own_reach(l), 0.0);
    }
}

// The rays leaving a water point x (normal n) for the light li: its shadow ray (glass tints it), then the reflected (branch 0)
// and refracted (branch 1) rays, each to the first thing it meets. Per branch: a point on its ray (where it landed; one metre
// along it where it met nothing), its state (0: no ray leaves here, 1: a ray that met nothing or carries no light, 2: it
// landed), and the light it carries as it leaves the wave (the light's intensity, a
// lamp's per steradian, or the sun's irradiance, times its share and the water's Fresnel split; the fog or the water on its way
// is taken where a point asks, over that point's distance: caustic_irradiance). Each ray leaves the wave a few millimetres
// along its own way: off its normal, the reflected light landed that much aside.
void trace_water(uint slot, uint li, float scale, float3 x, float3 n, out float3 land[2], out uint state[2], out float3 carried[2]) {
    land[0] = x; land[1] = x; state[0] = 0u; state[1] = 0u; carried[0] = 0.0; carried[1] = 0.0;
    float3 toLight, intensity; float tmax;
    light_at(li, x, toLight, intensity, tmax);
    const float cosIn = dot(n, toLight);
    if (!(cosIn > 1e-4)) return; // the light meets the wave's back: no ray leaves it here
    const float offset = 0.005;
    g_passWater = true; // the caustics' own rays: the level water below the waves is none of theirs
    const float3 light = intensity * scale * transmittance(x + toLight * offset, toLight, 0.0, tmax);
    // A corner in a shadow sends no ray on, but its beams keep their shape by the way its rays would go (the lit corners' light
    // ramps down to it: a shadow's edge on the water).
    const bool lit = any(light > 0.0);
    {
        const float reflected = fresnel_dielectric(cosIn, WATER_IOR);
        Hit hit; float3 tint;
        ray_cone(0.0, 0.0);
        // Reflected: wherever a wave sends it, down too (a wave's slope sends slanting light below the level), as far as it
        // surely stays above the water: to the waves' lowest point. What it meets before that (a rock at the shore) takes it;
        // past that it has met the water again, a second wave the tracer doesn't follow yet (its light is left out).
        const float3 r = reflect(-toLight, n);
        const float reach = r.y >= 0.0 ? g.params.y : max(x.y - (g.waterBottom[slot] - water_row(slot, 12).z), 0.0) / -r.y;
        land[0] = x + r; state[0] = 1u; carried[0] = r.y >= 0.0 ? light * reflected : 0.0;
        if (lit && trace_bounce(x + r * offset, r, reach, hit, tint)) { land[0] = hit.position; state[0] = 2u; carried[0] = light * reflected * tint; }
        // Refracted: under the water to the first thing it meets, however far.
        const float3 t = refract(-toLight, n, 1.0 / WATER_IOR);
        if (dot(t, t) > 1e-6) {
            land[1] = x + normalize(t); state[1] = 1u; carried[1] = light * (1.0 - reflected);
            if (lit && trace_bounce(x + t * offset, t, g.params.y, hit, tint)) { land[1] = hit.position; state[1] = 2u; carried[1] *= tint; }
        }
    }
    g_passWater = false;
}

// The solid angle a triangle fills seen from the origin (Van Oosterom and Strackee, 1983).
float solid_angle(float3 a, float3 b, float3 c) {
    const float la = length(a), lb = length(b), lc = length(c);
    const float volume = abs(dot(a, cross(b - a, c - a)));
    return 2.0 * atan2(volume, la * lb * lc + dot(a, b) * lc + dot(a, c) * lb + dot(b, c) * la);
}

// Whether a landed point lies in the camera's view or near it (half a view's width round it: the bed seen through the water lies
// a little outside the straight view).
bool in_view(float3 p) {
    const float4 c = mul(g.viewProj, float4(p - g.cameraPos.xyz, 1.0));
    return c.w > 0.0 && abs(c.x) <= 1.5 * c.w && abs(c.y) <= 1.5 * c.w;
}

// One beam (caustic_irradiance): a triangle of the water (corners a, b, c by their index in the tile, wound upwards as the grid
// gives them: (c - a) x (b - a) points up) and its three rays of one branch, with the light through it: a lamp's intensity times
// the solid angle the triangle fills seen from the lamp, the sun's irradiance times its area across the light, at each corner as
// that corner's ray carries it (the lamp's cone and cookie, the shadow, the Fresnel split vary across it; read across the beam
// by a point's weights, its mean is the triangle's light: none is made or lost). Its slab: from the nearest thing its rays met
// to the farthest, and past both by half its breadth there (the surface between three rays is no plane: a rock's bumps, a
// ledge) or a few pixels; a ray that met nothing reaches as far as the farthest that did. No beam is left out for where its
// rays went: one across an edge lights both sides of it, each point by its own ray back to the water.
void emit(uint branch, uint ia, uint ib, uint ic, uint li, uint slot) {
    const uint ta = gs_state[branch][ia], tb = gs_state[branch][ib], tc = gs_state[branch][ic];
    if (ta == 0u || tb == 0u || tc == 0u) return; // a corner off the water, or on a wave's back to the light
    const bool la = ta == 2u, lb = tb == 2u, lc = tc == 2u;
    if (!la && !lb && !lc) return; // the light goes on into the sky: nothing takes it
    const float3 qa = gs_carried[branch][ia], qb = gs_carried[branch][ib], qc = gs_carried[branch][ic];
    if (!any(qa + qb + qc > 0.0)) return;
    const float3 xa = gs_water[ia], xb = gs_water[ib], xc = gs_water[ic];
    const float3 up = cross(xc - xa, xb - xa); // twice the water triangle's area, along its upper side
    float share;
    if (li == SUN) share = max(0.0, 0.5 * dot(up, sun_direction()));
    else {
        const float3 y = light_point(load_light(li), li);
        share = dot(up, y - xa) > 0.0 ? solid_angle(xa - y, xb - y, xc - y) : 0.0;
    }
    if (!(share > 0.0)) return;
    float3 da = gs_land[branch][ia] - xa, db = gs_land[branch][ib] - xb, dc = gs_land[branch][ic] - xc;
    const float sa = length(da), sb = length(db), sc = length(dc);
    if (!(sa > 1e-6 && sb > 1e-6 && sc > 1e-6)) return;
    da /= sa; db /= sb; dc /= sc;
    const float nearest = min(la ? sa : 1e30, min(lb ? sb : 1e30, lc ? sc : 1e30)), farthest = max(la ? sa : 0.0, max(lb ? sb : 0.0, lc ? sc : 0.0));
    // A point on each ray: where it landed, or as far along as the farthest that did.
    const float3 pa = xa + da * (la ? sa : farthest), pb = xb + db * (lb ? sb : farthest), pc = xc + dc * (lc ? sc : farthest);
    const float breadth = max(length(pb - pa), max(length(pc - pa), length(pc - pb)));
    const float margin = max(4.0 * caustic_reach((pa + pb + pc) / 3.0), 0.5 * breadth);
    const float s0 = max(nearest - margin, 0.02), s1 = farthest + margin;
    const float3 lo = min(min(xa + da * s0, xb + db * s0), min(min(xc + dc * s0, xa + da * s1), min(xb + db * s1, xc + dc * s1)));
    const float3 hi = max(max(xa + da * s0, xb + db * s0), max(max(xc + dc * s0, xa + da * s1), max(xb + db * s1, xc + dc * s1)));
    const float3 ea = to_storage(qa * (share / PI)), eb = to_storage(qb * (share / PI)), ec = to_storage(qc * (share / PI));
    if (!all(isfinite(ea + eb + ec)) || !all(isfinite(lo + hi))) return;
    RWByteAddressBuffer light = ResourceDescriptorHeap[g.caustics.x];
    RWByteAddressBuffer beams = ResourceDescriptorHeap[asuint(g.causticGrid.y)];
    RWByteAddressBuffer boxes = ResourceDescriptorHeap[pass.a.y];
    RWByteAddressBuffer counters = ResourceDescriptorHeap[g.caustics.z];
    uint at;
    counters.InterlockedAdd(0, 1u, at);
    if (at >= g.caustics.w) return; // past the room this frame (counted: tracer.cpp makes more room for the next)
    const uint base = at * CAUSTIC_BEAM;
    beams.Store4(base, uint4(asuint(xa), asuint(s0)));
    beams.Store4(base + 16, uint4(asuint(xb), asuint(s1)));
    beams.Store4(base + 32, uint4(asuint(xc), branch | (slot << 1)));
    beams.Store4(base + 48, uint4(asuint(pa), 0u));
    beams.Store4(base + 64, uint4(asuint(pb), 0u));
    beams.Store4(base + 80, uint4(asuint(pc), 0u));
    light.Store4(at * CAUSTIC_LIGHT, uint4(asuint(ea), asuint(eb.x)));
    light.Store4(at * CAUSTIC_LIGHT + 16, uint4(asuint(eb.yz), asuint(ec.xy)));
    light.Store(at * CAUSTIC_LIGHT + 32, asuint(ec.z));
    boxes.Store4(at * CAUSTIC_BOX, uint4(asuint(lo), asuint(hi.x)));
    boxes.Store2(at * CAUSTIC_BOX + 16, asuint(hi.yz));
}

[RootSignature(PT_ROOT_SIGNATURE)]
[numthreads(16, 16, 1)]
void main(uint3 gid : SV_GroupID, uint3 tid : SV_GroupThreadID, uint index : SV_GroupIndex) {
    if (pass.a.x == 0u) {
        if (g.caustics.x == INVALID) return;
        RWByteAddressBuffer counters = ResourceDescriptorHeap[g.caustics.z];
        if (index < 16u) counters.Store(index * 4, 0u);
        return;
    }
    if (pass.a.x == 2u) {
        // The room past this frame's beams: inactive boxes (a NaN least x), so the build that takes the whole room leaves the
        // last frame's there out.
        if (g.caustics.x == INVALID) return;
        RWByteAddressBuffer boxes = ResourceDescriptorHeap[pass.a.y];
        RWByteAddressBuffer counters = ResourceDescriptorHeap[g.caustics.z];
        const uint i = gid.x * 256u + index;
        if (i < g.caustics.w && i >= counters.Load(0)) boxes.Store(i * CAUSTIC_BOX, 0x7FC00000u);
        return;
    }
    if (g.caustics.x == INVALID) return;
    const uint slot = gid.z % SLOTS, level = gid.z / SLOTS;
    if (level >= uint(g.causticGrid.z) || !water_slot(slot)) return;
    const float cell = caustic_cell(level);
    const float4 lv = g.causticLevels[level];
    const int2 first = int2(gid.xy) * int(CELLS);
    if (any(first >= int(GRID))) return;
    // The finer level's square is that level's: a tile wholly inside it has nothing to do here.
    const bool inner = lv.z > -1e8;
    const int2 innerLo = int2(lv.zw), innerHi = innerLo + int(GRID / 2);
    if (inner && all(first >= innerLo) && all(first + int(CELLS) <= innerHi)) return;
    // Nor one off the slot's water.
    const float2 lo = lv.xy + float2(first) * cell, hi = lo + float(CELLS) * cell;
    const float4 extent = water_row(slot, 13);
    const float margin = water_row(slot, 12).z;
    if (any(hi < extent.xy - margin) || any(lo > extent.zw + margin)) return;

    pick_lights(index, slot, lo, hi, (gid.z * 64u + gid.y) * 64u + gid.x);
    const int2 corner = first + int2(tid.xy);
    float3 x = 0.0, n = float3(0.0, 1.0, 0.0);
    const bool ok = all(corner <= int(GRID)) && water_point(slot, lv.xy + float2(corner) * cell, x, n);
    gs_water[index] = x;
    if (index == 0u) { gs_seen[0] = 0u; gs_seen[1] = 0u; }
    GroupMemoryBarrierWithGroupSync();
    // A tile that lies whole in this level's grid, clear of the finer level's square, may go in cells of 5 x 5 (below).
    const bool whole = all(first + int(CELLS) <= int(GRID)) && !(inner && all(first + int(CELLS) > innerLo) && all(first < innerHi));
    for (uint k = 0; k < PICKS; ++k) {
        const uint li = gs_light[k];
        if (li == CAUSTIC_NONE) continue;
        float3 land[2], carried[2]; uint state[2];
        land[0] = x; land[1] = x; state[0] = 0u; state[1] = 0u; carried[0] = 0.0; carried[1] = 0.0;
        if (ok) trace_water(slot, li, gs_scale[k], x, n, land, state, carried);
        [unroll] for (uint br = 0; br < 2; ++br) {
            gs_land[br][index] = land[br]; gs_state[br][index] = state[br]; gs_carried[br][index] = carried[br];
            if (state[br] == 2u && in_view(land[br])) InterlockedOr(gs_seen[br], 1u);
        }
        GroupMemoryBarrierWithGroupSync();
        for (uint branch = 0; branch < 2; ++branch) {
            // Where none of the tile's rays of this branch landed in or near the camera's view, its beams go in cells of 5 x 5
            // (25 times fewer, each as much wider, with the same light in all): what a mirror or a bounce sees of the caustics
            // out of view is all there, coarser. In view every cell is a beam.
            const uint stride = whole && gs_seen[branch] == 0u ? 5u : 1u;
            const bool emits = tid.x < CELLS && tid.y < CELLS && tid.x % stride == 0u && tid.y % stride == 0u &&
                               all(corner < int(GRID)) && !(inner && all(corner >= innerLo) && all(corner < innerHi));
            if (!emits) continue;
            emit(branch, index, index + stride, index + stride * (TILE + 1), li, slot);
            emit(branch, index, index + stride * (TILE + 1), index + stride * TILE, li, slot);
        }
        GroupMemoryBarrierWithGroupSync();
        if (index == 0u) { gs_seen[0] = 0u; gs_seen[1] = 0u; }
        GroupMemoryBarrierWithGroupSync();
    }
}
