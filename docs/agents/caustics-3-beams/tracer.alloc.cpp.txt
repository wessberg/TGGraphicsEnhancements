// The caustics' beams as the build of their acceleration structure takes them (caustic_irradiance): one geometry of boxes, the
// first `count` of the room's (unused ones inactive: pt_caustics mode 2).
D3D12_RAYTRACING_GEOMETRY_DESC caustic_geometry(uint32_t count) {
    D3D12_RAYTRACING_GEOMETRY_DESC g{}; g.Type=D3D12_RAYTRACING_GEOMETRY_TYPE_PROCEDURAL_PRIMITIVE_AABBS; g.Flags=D3D12_RAYTRACING_GEOMETRY_FLAG_NO_DUPLICATE_ANYHIT_INVOCATION;
    g.AABBs.AABBCount=count; g.AABBs.AABBs.StartAddress=st.causticBoxes.va(); g.AABBs.AABBs.StrideInBytes=kCausticBox;
    return g;
}
D3D12_BUILD_RAYTRACING_ACCELERATION_STRUCTURE_INPUTS caustic_inputs(const D3D12_RAYTRACING_GEOMETRY_DESC* geometry) {
    D3D12_BUILD_RAYTRACING_ACCELERATION_STRUCTURE_INPUTS in{}; in.Type=D3D12_RAYTRACING_ACCELERATION_STRUCTURE_TYPE_BOTTOM_LEVEL;
    in.Flags=D3D12_RAYTRACING_ACCELERATION_STRUCTURE_BUILD_FLAG_PREFER_FAST_BUILD; in.NumDescs=1; in.DescsLayout=D3D12_ELEMENTS_LAYOUT_ARRAY; in.pGeometryDescs=geometry;
    return in;
}
// Room for the caustics' beams (pt_caustics.hlsl): at least `wanted`, a power of two from half a million up, at most 4 million
// (past it, a frame's beams beyond the room light nothing, and the log says so); the old buffers are released once the frames
// that may read them are done. The top level (one instance) is made once.
void make_caustic_beams(uint32_t wanted) {
    uint32_t capacity=1u<<19; while(capacity<wanted && capacity<(1u<<22)) capacity<<=1;
    st.causticWanted=0;
    if(capacity==st.causticCapacity && st.causticBeams.resource) return;
    for(Buffer* b:{&st.causticBeams, &st.causticLight, &st.causticBoxes, &st.causticBlas, &st.causticBlasScratch}) if(b->resource) st.release.add(st.consumer.value, b->resource.Get());
    st.heap.release(st.causticBeamsUav); st.heap.release(st.causticLightUav); st.heap.release(st.causticBoxesUav);
    const auto uavFlag=D3D12_RESOURCE_FLAG_ALLOW_UNORDERED_ACCESS;
    st.causticBeams=make_buffer(st.d12.Get(), uint64_t(capacity)*kCausticBeam, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_UNORDERED_ACCESS, "Create the caustics' beams");
    st.causticLight=make_buffer(st.d12.Get(), uint64_t(capacity)*kCausticLight, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_UNORDERED_ACCESS, "Create the caustics' light");
    st.causticBoxes=make_buffer(st.d12.Get(), uint64_t(capacity)*kCausticBox, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_UNORDERED_ACCESS, "Create the caustics' boxes");
    st.causticBeamsUav=raw_uav(st.d12.Get(), st.heap, st.causticBeams.resource.Get(), uint64_t(capacity)*kCausticBeam);
    st.causticLightUav=raw_uav(st.d12.Get(), st.heap, st.causticLight.resource.Get(), uint64_t(capacity)*kCausticLight);
    st.causticBoxesUav=raw_uav(st.d12.Get(), st.heap, st.causticBoxes.resource.Get(), uint64_t(capacity)*kCausticBox);
    const D3D12_RAYTRACING_GEOMETRY_DESC geometry=caustic_geometry(capacity);
    const auto in=caustic_inputs(&geometry);
    D3D12_RAYTRACING_ACCELERATION_STRUCTURE_PREBUILD_INFO info{}; st.d12->GetRaytracingAccelerationStructurePrebuildInfo(&in, &info);
    st.causticBlas=make_buffer(st.d12.Get(), info.ResultDataMaxSizeInBytes, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_RAYTRACING_ACCELERATION_STRUCTURE, "Create the caustics' structure");
    st.causticBlasScratch=make_buffer(st.d12.Get(), info.ScratchDataSizeInBytes, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_UNORDERED_ACCESS, "Create the caustics' structure scratch");
    if(!st.causticTlas.resource) {
        D3D12_BUILD_RAYTRACING_ACCELERATION_STRUCTURE_INPUTS top{}; top.Type=D3D12_RAYTRACING_ACCELERATION_STRUCTURE_TYPE_TOP_LEVEL; top.NumDescs=1; top.DescsLayout=D3D12_ELEMENTS_LAYOUT_ARRAY;
        D3D12_RAYTRACING_ACCELERATION_STRUCTURE_PREBUILD_INFO topInfo{}; st.d12->GetRaytracingAccelerationStructurePrebuildInfo(&top, &topInfo);
        st.causticTlas=make_buffer(st.d12.Get(), topInfo.ResultDataMaxSizeInBytes, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_RAYTRACING_ACCELERATION_STRUCTURE, "Create the caustics' top level");
        st.causticTlasScratch=make_buffer(st.d12.Get(), topInfo.ScratchDataSizeInBytes, D3D12_HEAP_TYPE_DEFAULT, uavFlag, D3D12_RESOURCE_STATE_UNORDERED_ACCESS, "Create the caustics' top level scratch");
        st.causticTlasSrv=st.heap.allocate();
        D3D12_SHADER_RESOURCE_VIEW_DESC v{}; v.ViewDimension=D3D12_SRV_DIMENSION_RAYTRACING_ACCELERATION_STRUCTURE; v.Shader4ComponentMapping=D3D12_DEFAULT_SHADER_4_COMPONENT_MAPPING;
        v.RaytracingAccelerationStructure.Location=st.causticTlas.va();
        st.d12->CreateShaderResourceView(nullptr, &v, st.heap.cpu(st.causticTlasSrv));
    }
    log("The caustics' beams: room for "+std::to_string(capacity)+(st.causticCapacity ? " now (a frame wanted "+std::to_string(wanted)+(wanted>capacity ? "; the rest light nothing)" : ")") : std::string())+
        ", "+std::to_string((uint64_t(capacity)*(kCausticBeam+kCausticLight+kCausticBox)+info.ResultDataMaxSizeInBytes+info.ScratchDataSizeInBytes)>>20)+" MB with their acceleration structure");
    st.causticCapacity=capacity;
}
